# 案例 3：秒杀系统 - 高并发限流与防超卖设计

## 需求场景

秒杀系统中，需要处理高并发请求，要求：

-   限制用户请求频率（防止刷单）
-   限制总请求量（防止系统崩溃）
-   保证库存扣减的准确性（防止超卖）
-   支持秒杀活动预热
-   处理秒杀结果（成功/失败）

## 第一次设计（错误）

```javascript
// 错误示例：没有限流，直接操作数据库
async function seckill(userId, activityId) {
    // 检查用户是否已购买
    const purchased = await db.query(
        'SELECT * FROM seckill_orders WHERE user_id = ? AND activity_id = ?',
        [userId, activityId]
    );

    if (purchased) {
        throw new Error('您已参与过此活动');
    }

    // 检查库存
    const activity = await db.query(
        'SELECT stock FROM seckill_activities WHERE id = ?',
        [activityId]
    );
    if (activity.stock <= 0) {
        throw new Error('商品已售罄');
    }

    // 扣减库存
    await db.query(
        'UPDATE seckill_activities SET stock = stock - 1 WHERE id = ?',
        [activityId]
    );

    // 创建订单
    await db.query(
        'INSERT INTO seckill_orders (user_id, activity_id) VALUES (?, ?)',
        [userId, activityId]
    );

    return { success: true };
}
```

**问题 1**：没有限流，高并发下数据库直接崩溃
**问题 2**：库存扣减存在并发问题，可能超卖
**问题 3**：每个请求都查数据库，性能极差
**问题 4**：没有预热机制，活动开始瞬间压力巨大

## 设计思路

**核心问题**：

1. 如何限流？→ 使用 Redis 实现令牌桶或滑动窗口限流
2. 如何防止超卖？→ 使用 Redis 原子操作扣减库存
3. 如何提高性能？→ 使用 Redis 缓存活动信息，减少数据库访问
4. 如何防止重复购买？→ 使用 Redis Set 记录已购买用户

**解决方案**：

-   用户限流：Redis String + EXPIRE，key: `limit:user:{userId}:{activityId}`
-   全局限流：Redis String 计数器，key: `limit:global:{activityId}`
-   库存管理：Redis String，key: `stock:{activityId}`
-   已购买用户：Redis Set，key: `purchased:{activityId}`
-   秒杀结果：Redis Hash，key: `result:{userId}:{activityId}`

## 最终设计

### Redis 数据结构设计

```redis
# 用户限流（防止同一用户频繁请求）
limit:user:100:2001 -> "1"
TTL: 60秒（1分钟内只能请求1次）

# 全局限流计数器（限制总请求量）
limit:global:2001 -> 10000
# 每秒最多处理10000个请求

# 活动库存
stock:2001 -> 100
# 秒杀商品库存

# 已购买用户集合（防止重复购买）
purchased:2001 -> {100, 101, 102, ...}
# 已成功购买的用户ID集合

# 秒杀结果（Hash）
result:100:2001 -> {
  "status": "success",  # success/failed/sold_out
  "orderId": "20231201123456",
  "timestamp": "1701417600000"
}
TTL: 3600秒（1小时）
```

### 代码实现

```javascript
const redis = require('redis');
const client = redis.createClient();

// Lua 脚本：原子性秒杀操作
const seckillScript = `
  local stockKey = KEYS[1]
  local purchasedKey = KEYS[2]
  local resultKey = KEYS[3]
  local userId = ARGV[1]
  local activityId = ARGV[2]
  
  -- 检查是否已购买
  local isPurchased = redis.call('SISMEMBER', purchasedKey, userId)
  if isPurchased == 1 then
    return {0, '您已参与过此活动'}
  end
  
  -- 检查库存
  local stock = redis.call('GET', stockKey)
  if not stock then
    return {-1, '活动不存在'}
  end
  
  stock = tonumber(stock)
  if stock <= 0 then
    -- 记录结果：已售罄
    redis.call('HSET', resultKey, 'status', 'sold_out', 'timestamp', ARGV[3])
    redis.call('EXPIRE', resultKey, 3600)
    return {1, '商品已售罄'}
  end
  
  -- 扣减库存
  local newStock = redis.call('DECR', stockKey)
  
  -- 记录已购买用户
  redis.call('SADD', purchasedKey, userId)
  
  -- 记录结果：成功
  local orderId = 'SK' .. userId .. activityId .. ARGV[3]
  redis.call('HSET', resultKey, 'status', 'success', 'orderId', orderId, 'timestamp', ARGV[3])
  redis.call('EXPIRE', resultKey, 3600)
  
  return {2, orderId, newStock}
`;

/**
 * 检查用户限流
 */
async function checkUserLimit(userId, activityId) {
    const limitKey = `limit:user:${userId}:${activityId}`;
    const exists = await client.exists(limitKey);

    if (exists) {
        throw new Error('请求过于频繁，请稍后再试');
    }

    // 设置限流标记，60秒过期
    await client.setEx(limitKey, 60, '1');
    return true;
}

/**
 * 检查全局限流
 */
async function checkGlobalLimit(activityId, maxRequests = 10000) {
    const limitKey = `limit:global:${activityId}`;
    const current = await client.incr(limitKey);

    // 如果是第一次设置，设置过期时间为1秒
    if (current === 1) {
        await client.expire(limitKey, 1);
    }

    if (current > maxRequests) {
        throw new Error('系统繁忙，请稍后重试');
    }

    return true;
}

/**
 * 秒杀核心逻辑
 */
async function seckill(userId, activityId) {
    const timestamp = Date.now().toString();

    // 1. 用户限流检查
    try {
        await checkUserLimit(userId, activityId);
    } catch (error) {
        return { success: false, message: error.message };
    }

    // 2. 全局限流检查
    try {
        await checkGlobalLimit(activityId, 10000);
    } catch (error) {
        return { success: false, message: error.message };
    }

    // 3. 检查是否已购买（快速失败）
    const purchasedKey = `purchased:${activityId}`;
    const isPurchased = await client.sIsMember(purchasedKey, userId.toString());
    if (isPurchased) {
        return { success: false, message: '您已参与过此活动' };
    }

    // 4. 执行秒杀（Lua 脚本保证原子性）
    const stockKey = `stock:${activityId}`;
    const resultKey = `result:${userId}:${activityId}`;

    const result = await client.eval(seckillScript, {
        keys: [stockKey, purchasedKey, resultKey],
        arguments: [userId.toString(), activityId.toString(), timestamp],
    });

    const [status, ...args] = result;

    if (status === 0) {
        return { success: false, message: args[0] };
    } else if (status === 1) {
        return { success: false, message: args[0] };
    } else if (status === 2) {
        const [orderId, remainingStock] = args;

        // 5. 异步创建订单（最终一致性）
        createOrderInDb(userId, activityId, orderId).catch((err) => {
            console.error('创建订单失败:', err);
            // 可以记录到消息队列，后续重试
        });

        return {
            success: true,
            orderId: orderId,
            message: '秒杀成功',
            remainingStock: remainingStock,
        };
    } else {
        return { success: false, message: args[0] || '系统错误' };
    }
}

/**
 * 查询秒杀结果
 */
async function getSeckillResult(userId, activityId) {
    const resultKey = `result:${userId}:${activityId}`;
    const result = await client.hGetAll(resultKey);

    if (!result || Object.keys(result).length === 0) {
        return null;
    }

    return {
        status: result.status,
        orderId: result.orderId || null,
        timestamp: parseInt(result.timestamp),
    };
}

/**
 * 预热活动数据（活动开始前执行）
 */
async function warmUpSeckill(activityId, stock) {
    const stockKey = `stock:${activityId}`;
    await client.set(stockKey, stock.toString());

    // 清空已购买记录（新活动开始）
    const purchasedKey = `purchased:${activityId}`;
    await client.del(purchasedKey);

    console.log(`活动${activityId}预热完成，库存: ${stock}`);
}

/**
 * 获取活动剩余库存
 */
async function getRemainingStock(activityId) {
    const stockKey = `stock:${activityId}`;
    const stock = await client.get(stockKey);
    return stock ? parseInt(stock) : 0;
}

/**
 * 异步创建订单（最终一致性）
 */
async function createOrderInDb(userId, activityId, orderId) {
    await db.query(
        'INSERT INTO seckill_orders (order_id, user_id, activity_id, created_at) VALUES (?, ?, ?, NOW())',
        [orderId, userId, activityId]
    );

    // 更新数据库库存（最终一致性）
    await db.query(
        'UPDATE seckill_activities SET stock = GREATEST(stock - 1, 0) WHERE id = ?',
        [activityId]
    );
}

/**
 * 使用令牌桶算法限流（可选，更灵活的限流方案）
 */
async function checkTokenBucket(
    userId,
    activityId,
    capacity = 10,
    refillRate = 1
) {
    const bucketKey = `token:bucket:${userId}:${activityId}`;
    const lastRefillKey = `token:last:${userId}:${activityId}`;

    const now = Date.now();
    const lastRefill = (await client.get(lastRefillKey)) || 0;
    const elapsed = (now - parseInt(lastRefill)) / 1000; // 秒

    // 计算应该补充的令牌数
    const tokensToAdd = Math.floor(elapsed * refillRate);

    if (tokensToAdd > 0) {
        const currentTokens = (await client.get(bucketKey)) || 0;
        const newTokens = Math.min(
            parseInt(currentTokens) + tokensToAdd,
            capacity
        );
        await client.set(bucketKey, newTokens.toString());
        await client.set(lastRefillKey, now.toString());

        if (newTokens > 0) {
            await client.decr(bucketKey);
            return true;
        }
    } else {
        const tokens = await client.get(bucketKey);
        if (tokens && parseInt(tokens) > 0) {
            await client.decr(bucketKey);
            return true;
        }
    }

    throw new Error('请求过于频繁，请稍后再试');
}
```

## 设计要点

1. **多层限流**：用户限流 + 全局限流，防止系统过载
2. **原子性操作**：使用 Lua 脚本保证库存扣减、记录购买、更新结果的原子性
3. **快速失败**：在 Lua 脚本外先检查已购买状态，减少脚本执行次数
4. **防重复购买**：使用 Set 记录已购买用户，O(1) 时间复杂度判断
5. **结果缓存**：使用 Hash 存储秒杀结果，避免重复查询数据库
6. **预热机制**：活动开始前预热库存数据，减少数据库压力
7. **最终一致性**：先更新 Redis，再异步更新数据库

## 实际应用

```javascript
// 1. 活动开始前预热
await warmUpSeckill(2001, 100); // 预热100件库存

// 2. 用户发起秒杀请求
const result = await seckill(100, 2001);
if (result.success) {
    console.log(
        `秒杀成功！订单号: ${result.orderId}，剩余库存: ${result.remainingStock}`
    );
} else {
    console.log(`秒杀失败: ${result.message}`);
}

// 3. 查询秒杀结果
const seckillResult = await getSeckillResult(100, 2001);
if (seckillResult) {
    console.log(
        `秒杀状态: ${seckillResult.status}，订单号: ${seckillResult.orderId}`
    );
}

// 4. 查询剩余库存
const remaining = await getRemainingStock(2001);
console.log(`活动2001剩余库存: ${remaining}`);
```

# 案例 1：电商系统 - 商品缓存与库存扣减设计

## 需求场景

电商系统中，商品信息查询频繁，需要：

-   商品详情查询（商品名称、价格、图片等）
-   商品库存查询和扣减
-   支持高并发访问（QPS 10000+）
-   保证库存扣减的准确性，防止超卖

## 第一次设计（错误）

```javascript
// 错误示例：直接查询数据库
async function getProduct(productId) {
    // 每次都查数据库，性能差
    const product = await db.query('SELECT * FROM products WHERE id = ?', [
        productId,
    ]);
    return product;
}

async function reduceStock(productId, quantity) {
    // 并发情况下可能超卖
    const product = await db.query('SELECT stock FROM products WHERE id = ?', [
        productId,
    ]);
    if (product.stock < quantity) {
        throw new Error('库存不足');
    }
    await db.query('UPDATE products SET stock = stock - ? WHERE id = ?', [
        quantity,
        productId,
    ]);
}
```

**问题 1**：每次查询都访问数据库，QPS 高时数据库压力大
**问题 2**：库存扣减存在并发问题，可能出现超卖
**问题 3**：没有缓存，响应时间长

## 设计思路

**核心问题**：

1. 如何减少数据库压力？→ 使用 Redis 缓存商品信息
2. 如何防止缓存穿透？→ 缓存空值，使用布隆过滤器
3. 如何防止缓存击穿？→ 使用分布式锁
4. 如何防止缓存雪崩？→ 设置随机过期时间
5. 如何保证库存扣减准确性？→ 使用 Redis 原子操作或 Lua 脚本

**解决方案**：

-   商品信息缓存：Redis String 结构，key: `product:{id}`
-   库存缓存：Redis String 结构，key: `stock:{id}`
-   使用 Lua 脚本保证库存扣减的原子性
-   缓存更新策略：先更新数据库，再删除缓存（Cache Aside 模式）

## 最终设计

### Redis 数据结构设计

```redis
# 商品信息缓存（JSON 格式）
product:1001 -> {
  "id": 1001,
  "name": "iPhone 15 Pro",
  "price": 7999.00,
  "image": "https://example.com/iphone15.jpg",
  "description": "..."
}
TTL: 3600秒（1小时）

# 商品库存缓存（数字）
stock:1001 -> 100
TTL: 1800秒（30分钟）

# 商品库存锁（防止缓存击穿）
lock:stock:1001 -> "locked"
TTL: 10秒
```

### 代码实现

```javascript
const redis = require('redis');
const client = redis.createClient();

// Lua 脚本：原子性扣减库存
const reduceStockScript = `
  local stock = redis.call('GET', KEYS[1])
  if not stock then
    return {-1, '库存信息不存在'}
  end
  stock = tonumber(stock)
  local quantity = tonumber(ARGV[1])
  if stock < quantity then
    return {0, '库存不足'}
  end
  local newStock = stock - quantity
  redis.call('SET', KEYS[1], newStock)
  return {1, newStock}
`;

/**
 * 获取商品信息（带缓存）
 */
async function getProduct(productId) {
    const cacheKey = `product:${productId}`;

    // 1. 先查缓存
    let product = await client.get(cacheKey);
    if (product) {
        return JSON.parse(product);
    }

    // 2. 缓存未命中，查数据库
    const productFromDb = await db.query(
        'SELECT * FROM products WHERE id = ?',
        [productId]
    );

    // 3. 防止缓存穿透：即使商品不存在也缓存空值
    if (!productFromDb) {
        await client.setEx(cacheKey, 300, JSON.stringify(null)); // 缓存5分钟
        return null;
    }

    // 4. 写入缓存，设置随机过期时间（防止缓存雪崩）
    const ttl = 3600 + Math.floor(Math.random() * 600); // 3600-4200秒
    await client.setEx(cacheKey, ttl, JSON.stringify(productFromDb));

    return productFromDb;
}

/**
 * 扣减库存（原子性操作）
 */
async function reduceStock(productId, quantity) {
    const stockKey = `stock:${productId}`;
    const lockKey = `lock:stock:${productId}`;

    // 1. 获取分布式锁（防止缓存击穿）
    const lockAcquired = await client.set(lockKey, 'locked', {
        EX: 10, // 10秒过期
        NX: true, // 只在key不存在时设置
    });

    if (!lockAcquired) {
        throw new Error('系统繁忙，请稍后重试');
    }

    try {
        // 2. 检查库存缓存是否存在
        let stock = await client.get(stockKey);

        if (!stock) {
            // 缓存未命中，从数据库加载
            const product = await db.query(
                'SELECT stock FROM products WHERE id = ?',
                [productId]
            );
            if (!product) {
                throw new Error('商品不存在');
            }
            stock = product.stock;
            // 写入缓存
            await client.setEx(stockKey, 1800, stock.toString());
        }

        // 3. 使用 Lua 脚本原子性扣减库存
        const result = await client.eval(reduceStockScript, {
            keys: [stockKey],
            arguments: [quantity.toString()],
        });

        const [status, message] = result;

        if (status === -1) {
            throw new Error('库存信息不存在');
        }
        if (status === 0) {
            throw new Error('库存不足');
        }

        // 4. 异步更新数据库（最终一致性）
        updateStockInDb(productId, message).catch((err) => {
            console.error('更新数据库库存失败:', err);
            // 可以记录到消息队列，后续重试
        });

        return {
            success: true,
            remainingStock: message,
        };
    } finally {
        // 5. 释放锁
        await client.del(lockKey);
    }
}

/**
 * 异步更新数据库库存
 */
async function updateStockInDb(productId, newStock) {
    await db.query('UPDATE products SET stock = ? WHERE id = ?', [
        newStock,
        productId,
    ]);
}

/**
 * 预热缓存（系统启动时或定时任务）
 */
async function warmUpCache(productIds) {
    for (const productId of productIds) {
        const product = await db.query('SELECT * FROM products WHERE id = ?', [
            productId,
        ]);
        if (product) {
            await client.setEx(
                `product:${productId}`,
                3600,
                JSON.stringify(product)
            );
            await client.setEx(
                `stock:${productId}`,
                1800,
                product.stock.toString()
            );
        }
    }
}
```

## 真实数据示例

```redis
# 商品信息缓存
product:1001 -> {"id":1001,"name":"iPhone 15 Pro","price":7999.00,"image":"https://example.com/iphone15.jpg"}
product:1002 -> {"id":1002,"name":"MacBook Pro","price":12999.00,"image":"https://example.com/macbook.jpg"}

# 库存缓存
stock:1001 -> 50
stock:1002 -> 30

# 库存锁（扣减时临时存在）
lock:stock:1001 -> "locked"  # 10秒后自动过期
```

## 设计要点

1. **缓存穿透防护**：商品不存在时也缓存空值，避免频繁查询数据库
2. **缓存击穿防护**：使用分布式锁，防止热点 key 失效时大量请求打到数据库
3. **缓存雪崩防护**：设置随机过期时间，避免大量 key 同时失效
4. **原子性保证**：使用 Lua 脚本保证库存扣减的原子性，防止超卖
5. **最终一致性**：先更新 Redis，再异步更新数据库，保证高性能
6. **缓存更新策略**：采用 Cache Aside 模式，先更新数据库，再删除缓存

## 实际应用

```javascript
// 1. 查询商品详情
const product = await getProduct(1001);
console.log(product.name); // "iPhone 15 Pro"

// 2. 扣减库存
try {
    const result = await reduceStock(1001, 2);
    console.log(`扣减成功，剩余库存: ${result.remainingStock}`);
} catch (error) {
    console.error(error.message); // "库存不足" 或 "系统繁忙，请稍后重试"
}

// 3. 系统启动时预热热点商品缓存
await warmUpCache([1001, 1002, 1003]);
```

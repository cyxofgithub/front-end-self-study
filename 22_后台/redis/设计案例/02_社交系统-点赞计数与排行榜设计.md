# 案例 2：社交系统 - 点赞计数与排行榜设计

## 需求场景

社交系统中，用户可以对文章、评论进行点赞，需要支持：

-   点赞/取消点赞操作
-   查询点赞总数
-   查询用户是否已点赞
-   查询点赞用户列表（分页）
-   文章点赞排行榜（Top N）
-   支持高并发点赞（QPS 5000+）

## 第一次设计（错误）

```javascript
// 错误示例：使用数据库存储点赞关系
async function likeArticle(userId, articleId) {
    // 检查是否已点赞
    const exists = await db.query(
        'SELECT * FROM likes WHERE user_id = ? AND article_id = ?',
        [userId, articleId]
    );

    if (exists) {
        // 取消点赞
        await db.query(
            'DELETE FROM likes WHERE user_id = ? AND article_id = ?',
            [userId, articleId]
        );
        await db.query(
            'UPDATE articles SET like_count = like_count - 1 WHERE id = ?',
            [articleId]
        );
    } else {
        // 点赞
        await db.query(
            'INSERT INTO likes (user_id, article_id) VALUES (?, ?)',
            [userId, articleId]
        );
        await db.query(
            'UPDATE articles SET like_count = like_count + 1 WHERE id = ?',
            [articleId]
        );
    }
}

async function getLikeCount(articleId) {
    // 每次都查数据库统计
    const result = await db.query(
        'SELECT COUNT(*) as count FROM likes WHERE article_id = ?',
        [articleId]
    );
    return result.count;
}
```

**问题 1**：每次点赞都要写数据库，高并发下数据库压力大
**问题 2**：统计点赞数需要 COUNT 查询，性能差
**问题 3**：排行榜需要 ORDER BY + LIMIT，数据库压力大
**问题 4**：并发点赞时，like_count 可能不准确

## 设计思路

**核心问题**：

1. 如何高效存储点赞关系？→ 使用 Redis Set 存储点赞用户集合
2. 如何快速统计点赞数？→ 使用 Set 的 SCARD 命令
3. 如何实现排行榜？→ 使用 Redis Sorted Set（有序集合）
4. 如何保证点赞操作的原子性？→ 使用 Redis 事务或 Lua 脚本

**解决方案**：

-   点赞关系：Redis Set，key: `article:like:{articleId}`，存储用户 ID
-   点赞计数：使用 Set 的 SCARD 命令，O(1) 时间复杂度
-   排行榜：Redis Sorted Set，key: `article:like:rank`，score 为点赞数
-   用户点赞记录：Redis Set，key: `user:like:{userId}`，存储文章 ID（可选）

## 最终设计

### Redis 数据结构设计

```redis
# 文章点赞用户集合（Set）
article:like:1001 -> {100, 101, 102, 103, ...}
# 使用 SCARD 获取点赞数：O(1)
# 使用 SISMEMBER 判断是否点赞：O(1)

# 文章点赞排行榜（Sorted Set）
article:like:rank -> {
  "1001": 1500,  # 文章ID: 点赞数
  "1002": 1200,
  "1003": 980,
  ...
}
# 使用 ZREVRANGE 获取 Top N：O(log(N) + M)，N为总数，M为返回数量

# 用户点赞的文章集合（Set，可选）
user:like:100 -> {1001, 1002, 1003, ...}
# 用于查询用户点赞了哪些文章
```

### 代码实现

```javascript
const redis = require('redis');
const client = redis.createClient();

// Lua 脚本：原子性点赞/取消点赞
const toggleLikeScript = `
  local articleKey = KEYS[1]
  local rankKey = KEYS[2]
  local articleId = ARGV[1]
  local userId = ARGV[2]
  
  -- 检查是否已点赞
  local isMember = redis.call('SISMEMBER', articleKey, userId)
  
  if isMember == 1 then
    -- 取消点赞
    redis.call('SREM', articleKey, userId)
    local count = redis.call('SCARD', articleKey)
    redis.call('ZADD', rankKey, count, articleId)
    return {0, count}  -- 0表示取消点赞
  else
    -- 点赞
    redis.call('SADD', articleKey, userId)
    local count = redis.call('SCARD', articleKey)
    redis.call('ZADD', rankKey, count, articleId)
    return {1, count}  -- 1表示点赞成功
  end
`;

/**
 * 点赞/取消点赞文章
 */
async function toggleLikeArticle(userId, articleId) {
    const articleLikeKey = `article:like:${articleId}`;
    const rankKey = 'article:like:rank';

    // 使用 Lua 脚本保证原子性
    const result = await client.eval(toggleLikeScript, {
        keys: [articleLikeKey, rankKey],
        arguments: [articleId.toString(), userId.toString()],
    });

    const [action, count] = result;

    // 可选：更新用户点赞记录
    if (action === 1) {
        await client.sAdd(`user:like:${userId}`, articleId.toString());
    } else {
        await client.sRem(`user:like:${userId}`, articleId.toString());
    }

    // 异步更新数据库（最终一致性）
    updateLikeInDb(userId, articleId, action === 1).catch((err) => {
        console.error('更新数据库点赞失败:', err);
    });

    return {
        liked: action === 1,
        likeCount: count,
    };
}

/**
 * 获取文章点赞数
 */
async function getLikeCount(articleId) {
    const articleLikeKey = `article:like:${articleId}`;
    const count = await client.sCard(articleLikeKey);
    return count;
}

/**
 * 判断用户是否已点赞
 */
async function isLiked(userId, articleId) {
    const articleLikeKey = `article:like:${articleId}`;
    const result = await client.sIsMember(articleLikeKey, userId.toString());
    return result === 1;
}

/**
 * 获取文章点赞用户列表（分页）
 */
async function getLikeUsers(articleId, page = 1, pageSize = 20) {
    const articleLikeKey = `article:like:${articleId}`;
    const start = (page - 1) * pageSize;
    const end = start + pageSize - 1;

    // 使用 SRANDMEMBER 随机获取，或使用 SMEMBERS 获取全部后分页
    // 如果需要按时间排序，需要额外维护一个有序集合
    const userIds = await client.sRandMember(articleLikeKey, pageSize);

    return userIds.map((id) => parseInt(id));
}

/**
 * 获取文章点赞排行榜（Top N）
 */
async function getLikeRank(topN = 10) {
    const rankKey = 'article:like:rank';

    // ZREVRANGE 按分数降序排列，获取 Top N
    const results = await client.zRevRangeWithScores(rankKey, 0, topN - 1);

    return results.map((item) => ({
        articleId: parseInt(item.value),
        likeCount: item.score,
    }));
}

/**
 * 获取用户点赞的文章列表
 */
async function getUserLikedArticles(userId, page = 1, pageSize = 20) {
    const userLikeKey = `user:like:${userId}`;
    const start = (page - 1) * pageSize;
    const end = start + pageSize - 1;

    // 注意：Set 是无序的，如果需要按时间排序，需要额外维护有序集合
    const articleIds = await client.sMembers(userLikeKey);

    // 手动分页
    const paginatedIds = articleIds.slice(start, end + 1);

    return paginatedIds.map((id) => parseInt(id));
}

/**
 * 异步更新数据库（最终一致性）
 */
async function updateLikeInDb(userId, articleId, isLike) {
    if (isLike) {
        // 点赞：插入关系表，更新计数
        await db.query(
            'INSERT IGNORE INTO likes (user_id, article_id) VALUES (?, ?)',
            [userId, articleId]
        );
        await db.query(
            'UPDATE articles SET like_count = like_count + 1 WHERE id = ?',
            [articleId]
        );
    } else {
        // 取消点赞：删除关系表，更新计数
        await db.query(
            'DELETE FROM likes WHERE user_id = ? AND article_id = ?',
            [userId, articleId]
        );
        await db.query(
            'UPDATE articles SET like_count = GREATEST(like_count - 1, 0) WHERE id = ?',
            [articleId]
        );
    }
}

/**
 * 数据同步：从数据库加载点赞数据到 Redis（系统启动时或定时任务）
 */
async function syncLikesFromDb(articleId) {
    // 从数据库加载点赞用户列表
    const likes = await db.query(
        'SELECT user_id FROM likes WHERE article_id = ?',
        [articleId]
    );
    const userIds = likes.map((like) => like.user_id.toString());

    if (userIds.length > 0) {
        const articleLikeKey = `article:like:${articleId}`;
        await client.sAdd(articleLikeKey, userIds);

        // 更新排行榜
        await client.zAdd('article:like:rank', {
            score: userIds.length,
            value: articleId.toString(),
        });
    }
}
```

## 真实数据示例

```redis
# 文章1001的点赞用户集合
article:like:1001 -> {100, 101, 102, 103, 104}
# SCARD article:like:1001 -> 5（点赞数）

# 文章1002的点赞用户集合
article:like:1002 -> {100, 105, 106}
# SCARD article:like:1002 -> 3

# 点赞排行榜（按点赞数降序）
article:like:rank -> {
  "1001": 5,
  "1002": 3,
  "1003": 8
}
# ZREVRANGE article:like:rank 0 9 -> 获取 Top 10

# 用户100点赞的文章集合
user:like:100 -> {1001, 1002, 1003}
```

## 设计要点

1. **Set 存储点赞关系**：O(1) 时间复杂度判断是否点赞，O(1) 获取点赞数
2. **Sorted Set 实现排行榜**：使用有序集合维护排行榜，支持高效的 Top N 查询
3. **Lua 脚本保证原子性**：点赞和更新排行榜在一个原子操作中完成
4. **最终一致性**：先更新 Redis，再异步更新数据库
5. **分页查询**：使用 SRANDMEMBER 或 SMEMBERS 实现分页（注意 Set 无序特性）
6. **数据同步**：系统启动时从数据库加载热点数据到 Redis

## 实际应用

```javascript
// 1. 点赞文章
const result = await toggleLikeArticle(100, 1001);
console.log(
    `点赞${result.liked ? '成功' : '已取消'}，当前点赞数: ${result.likeCount}`
);

// 2. 查询点赞数
const count = await getLikeCount(1001);
console.log(`文章1001的点赞数: ${count}`);

// 3. 判断是否已点赞
const liked = await isLiked(100, 1001);
console.log(`用户100是否点赞了文章1001: ${liked}`);

// 4. 获取点赞排行榜 Top 10
const rank = await getLikeRank(10);
rank.forEach((item, index) => {
    console.log(`${index + 1}. 文章${item.articleId}: ${item.likeCount}个赞`);
});

// 5. 获取用户点赞的文章列表
const userLikes = await getUserLikedArticles(100, 1, 20);
console.log(`用户100点赞的文章:`, userLikes);
```

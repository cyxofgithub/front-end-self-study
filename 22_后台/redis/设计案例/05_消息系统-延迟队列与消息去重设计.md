# 案例 5：消息系统 - 延迟队列与消息去重设计

## 需求场景

消息系统中，需要实现以下功能：

-   延迟消息（指定时间后执行）
-   消息去重（防止重复处理）
-   消息优先级（高优先级消息优先处理）
-   消息重试（失败后自动重试）
-   消息持久化（防止丢失）
-   支持大量消息（百万级）

## 第一次设计（错误）

```javascript
// 错误示例：使用数据库轮询实现延迟消息
async function sendDelayedMessage(message, delaySeconds) {
    const executeTime = Date.now() + delaySeconds * 1000;
    await db.query(
        'INSERT INTO delayed_messages (message, execute_time, status) VALUES (?, ?, ?)',
        [JSON.stringify(message), executeTime, 'pending']
    );
}

// 定时任务：每秒查询一次
setInterval(async () => {
    const messages = await db.query(
        'SELECT * FROM delayed_messages WHERE execute_time <= ? AND status = ? LIMIT 100',
        [Date.now(), 'pending']
    );

    for (const msg of messages) {
        await processMessage(msg);
        await db.query('UPDATE delayed_messages SET status = ? WHERE id = ?', [
            'processed',
            msg.id,
        ]);
    }
}, 1000);
```

**问题 1**：数据库轮询性能差，延迟不精确
**问题 2**：没有消息去重机制，可能重复处理
**问题 3**：不支持优先级，所有消息平等处理
**问题 4**：高并发下数据库压力大

## 设计思路

**核心问题**：

1. 如何实现延迟消息？→ 使用 Redis Sorted Set，score 为执行时间
2. 如何实现消息去重？→ 使用 Redis Set 存储已处理消息的唯一标识
3. 如何实现优先级？→ 使用多个 Sorted Set 或调整 score 计算方式
4. 如何保证消息不丢失？→ 使用持久化 + 消息确认机制

**解决方案**：

-   延迟队列：Redis Sorted Set，key: `delay:queue`，score 为执行时间戳
-   消息去重：Redis Set，key: `processed:messages`，存储消息唯一标识
-   优先级队列：多个 Sorted Set，key: `delay:queue:high`、`delay:queue:normal`、`delay:queue:low`
-   消息数据：Redis Hash，key: `message:{messageId}`，存储消息详情
-   重试队列：Redis List，key: `retry:queue`，存储失败的消息

## 最终设计

### Redis 数据结构设计

```redis
# 延迟队列（Sorted Set，按执行时间排序）
delay:queue:high -> {
  "msg:1001": 1701417600000,  # 消息ID: 执行时间戳（高优先级）
  "msg:1002": 1701417700000,
  ...
}
delay:queue:normal -> {
  "msg:2001": 1701417600000,  # 普通优先级
  ...
}
delay:queue:low -> {
  "msg:3001": 1701417600000,  # 低优先级
  ...
}

# 消息数据（Hash）
message:msg:1001 -> {
  "id": "msg:1001",
  "type": "email",
  "payload": "{\"to\":\"user@example.com\",\"subject\":\"Hello\"}",
  "priority": "high",
  "retryCount": "0",
  "maxRetries": "3"
}
TTL: 86400秒（24小时）

# 已处理消息集合（Set，用于去重）
processed:messages -> {msg:1001, msg:1002, ...}
# 存储已处理消息的唯一标识

# 重试队列（List）
retry:queue -> [msg:1001, msg:1002, ...]
# 存储需要重试的消息ID
```

### 代码实现

```javascript
const redis = require('redis');
const crypto = require('crypto');
const client = redis.createClient();

const MESSAGE_TTL = 86400; // 24小时
const PROCESSED_TTL = 86400 * 7; // 7天
const MAX_RETRIES = 3;
const RETRY_DELAY = 60; // 60秒后重试

/**
 * 生成消息唯一标识
 */
function generateMessageId(type, payload) {
    const hash = crypto
        .createHash('md5')
        .update(type + JSON.stringify(payload))
        .digest('hex');
    return `msg:${hash.substring(0, 16)}`;
}

/**
 * 发送延迟消息
 */
async function sendDelayedMessage(message, delaySeconds, priority = 'normal') {
    const messageId =
        message.id || generateMessageId(message.type, message.payload);
    const executeTime = Date.now() + delaySeconds * 1000;

    const queueKey = `delay:queue:${priority}`;
    const messageKey = `message:${messageId}`;

    // 检查是否已存在（去重）
    const exists = await client.exists(messageKey);
    if (exists) {
        throw new Error('消息已存在，请勿重复发送');
    }

    const messageData = {
        id: messageId,
        type: message.type,
        payload: JSON.stringify(message.payload),
        priority: priority,
        retryCount: '0',
        maxRetries: (message.maxRetries || MAX_RETRIES).toString(),
        createdAt: Date.now().toString(),
        executeTime: executeTime.toString(),
    };

    const multi = client.multi();

    // 1. 添加到延迟队列
    multi.zAdd(queueKey, {
        score: executeTime,
        value: messageId,
    });

    // 2. 存储消息数据
    multi.setEx(messageKey, MESSAGE_TTL, JSON.stringify(messageData));

    await multi.exec();

    return messageId;
}

/**
 * 处理延迟消息（定时任务调用）
 */
async function processDelayedMessages() {
    const now = Date.now();
    const priorities = ['high', 'normal', 'low'];

    for (const priority of priorities) {
        const queueKey = `delay:queue:${priority}`;

        // 获取到期的消息（score <= now）
        const messageIds = await client.zRangeByScore(queueKey, 0, now, {
            LIMIT: {
                offset: 0,
                count: 100,
            },
        });

        if (messageIds.length === 0) {
            continue;
        }

        for (const messageId of messageIds) {
            await processMessage(messageId, priority);
        }
    }
}

/**
 * 处理单条消息
 */
async function processMessage(messageId, priority) {
    const messageKey = `message:${messageId}`;
    const processedKey = 'processed:messages';
    const queueKey = `delay:queue:${priority}`;

    // 1. 检查是否已处理（去重）
    const isProcessed = await client.sIsMember(processedKey, messageId);
    if (isProcessed) {
        // 已处理，从队列中移除
        await client.zRem(queueKey, messageId);
        await client.del(messageKey);
        return { success: false, reason: 'already_processed' };
    }

    // 2. 获取消息数据
    const messageDataStr = await client.get(messageKey);
    if (!messageDataStr) {
        // 消息不存在，从队列中移除
        await client.zRem(queueKey, messageId);
        return { success: false, reason: 'message_not_found' };
    }

    const messageData = JSON.parse(messageDataStr);

    try {
        // 3. 处理消息（根据类型调用不同的处理函数）
        await handleMessage(messageData);

        // 4. 标记为已处理
        const multi = client.multi();
        multi.sAdd(processedKey, messageId);
        multi.expire(processedKey, PROCESSED_TTL);
        multi.zRem(queueKey, messageId);
        multi.del(messageKey);
        await multi.exec();

        return { success: true };
    } catch (error) {
        console.error(`处理消息失败: ${messageId}`, error);

        // 5. 处理失败，检查是否需要重试
        const retryCount = parseInt(messageData.retryCount) || 0;
        const maxRetries = parseInt(messageData.maxRetries) || MAX_RETRIES;

        if (retryCount < maxRetries) {
            // 添加到重试队列
            await addToRetryQueue(messageId, messageData, priority);
        } else {
            // 超过最大重试次数，标记为失败
            await markMessageFailed(messageId, queueKey, error.message);
        }

        return {
            success: false,
            reason: 'processing_failed',
            error: error.message,
        };
    }
}

/**
 * 处理消息（根据类型）
 */
async function handleMessage(messageData) {
    const payload = JSON.parse(messageData.payload);

    switch (messageData.type) {
        case 'email':
            await sendEmail(payload);
            break;
        case 'sms':
            await sendSms(payload);
            break;
        case 'push':
            await sendPushNotification(payload);
            break;
        default:
            throw new Error(`未知的消息类型: ${messageData.type}`);
    }
}

/**
 * 添加到重试队列
 */
async function addToRetryQueue(messageId, messageData, priority) {
    const retryQueueKey = 'retry:queue';
    const retryTime = Date.now() + RETRY_DELAY * 1000;

    // 更新重试次数
    messageData.retryCount = (parseInt(messageData.retryCount) + 1).toString();
    messageData.executeTime = retryTime.toString();

    const multi = client.multi();

    // 1. 更新消息数据
    multi.setEx(
        `message:${messageId}`,
        MESSAGE_TTL,
        JSON.stringify(messageData)
    );

    // 2. 添加到重试队列（使用延迟队列）
    multi.zAdd(`delay:queue:${priority}`, {
        score: retryTime,
        value: messageId,
    });

    await multi.exec();
}

/**
 * 标记消息失败
 */
async function markMessageFailed(messageId, queueKey, errorMessage) {
    const failedKey = `failed:message:${messageId}`;
    await client.setEx(
        failedKey,
        MESSAGE_TTL,
        JSON.stringify({
            messageId: messageId,
            error: errorMessage,
            failedAt: Date.now(),
        })
    );

    await client.zRem(queueKey, messageId);
    await client.del(`message:${messageId}`);
}

/**
 * 查询消息状态
 */
async function getMessageStatus(messageId) {
    const messageKey = `message:${messageId}`;
    const processedKey = 'processed:messages';
    const failedKey = `failed:message:${messageId}`;

    // 检查是否已处理
    const isProcessed = await client.sIsMember(processedKey, messageId);
    if (isProcessed) {
        return { status: 'processed' };
    }

    // 检查是否失败
    const failedData = await client.get(failedKey);
    if (failedData) {
        return { status: 'failed', ...JSON.parse(failedData) };
    }

    // 检查是否在队列中
    const messageData = await client.get(messageKey);
    if (messageData) {
        const data = JSON.parse(messageData);
        const executeTime = parseInt(data.executeTime);
        const now = Date.now();

        return {
            status: executeTime <= now ? 'pending' : 'scheduled',
            executeTime: executeTime,
            retryCount: parseInt(data.retryCount),
            maxRetries: parseInt(data.maxRetries),
        };
    }

    return { status: 'not_found' };
}

/**
 * 取消延迟消息
 */
async function cancelDelayedMessage(messageId) {
    const messageKey = `message:${messageId}`;
    const messageDataStr = await client.get(messageKey);

    if (!messageDataStr) {
        return false;
    }

    const messageData = JSON.parse(messageDataStr);
    const queueKey = `delay:queue:${messageData.priority}`;

    const multi = client.multi();
    multi.zRem(queueKey, messageId);
    multi.del(messageKey);
    await multi.exec();

    return true;
}

/**
 * 获取队列统计信息
 */
async function getQueueStats() {
    const priorities = ['high', 'normal', 'low'];
    const stats = {};

    for (const priority of priorities) {
        const queueKey = `delay:queue:${priority}`;
        const count = await client.zCard(queueKey);
        const nextExecuteTime = await client.zRange(queueKey, 0, 0, {
            REV: false,
            WITHSCORES: true,
        });

        stats[priority] = {
            count: count,
            nextExecuteTime:
                nextExecuteTime.length > 0
                    ? parseInt(nextExecuteTime[0].score)
                    : null,
        };
    }

    const processedCount = await client.sCard('processed:messages');
    const retryCount =
        (await client.zCard('delay:queue:high')) +
        (await client.zCard('delay:queue:normal')) +
        (await client.zCard('delay:queue:low'));

    return {
        queues: stats,
        processedCount: processedCount,
        pendingCount: retryCount,
    };
}

// 定时任务：每秒处理一次延迟消息
setInterval(() => {
    processDelayedMessages().catch((err) => {
        console.error('处理延迟消息失败:', err);
    });
}, 1000);
```

## 设计要点

1. **Sorted Set 实现延迟队列**：使用 score 存储执行时间，支持精确的延迟执行
2. **多优先级队列**：使用多个 Sorted Set 实现优先级，高优先级消息优先处理
3. **消息去重**：使用 Set 存储已处理消息，防止重复处理
4. **消息持久化**：使用 Hash 存储消息详情，设置 TTL 防止内存泄漏
5. **重试机制**：失败消息重新加入延迟队列，支持指数退避
6. **定时处理**：定时任务扫描到期消息，按优先级顺序处理
7. **消息状态查询**：支持查询消息的处理状态（待处理/已处理/失败）

```javascript
<script>
(function() {
  const errorQueue = new Map();
  const REPORT_URL = '/api/log/report';
  const RATE_LIMIT = 10000; // 10s限流

  function report(data) {
    const key = data.type + data.message.slice(0, 100);
    const last = errorQueue.get(key) || 0;
    if (Date.now() - last < RATE_LIMIT) return;
    errorQueue.set(key, Date.now());

    const body = JSON.stringify({
      ...data,
      href: location.href,
      ua: navigator.userAgent,
      time: Date.now()
    });
    navigator.sendBeacon?.(REPORT_URL, body) || fetch(REPORT_URL, {
      method: 'POST', body, keepalive: true
    });
  }

  // 1. 同步JS、定时器错误
  window.onerror = function(msg, url, line, col, err) {
    if (!err || msg === 'Script error.') return true;
    report({ type: 'js_error', msg, url, line, col, stack: err.stack });
    return true;
  };

  // 2. Promise异常
  window.addEventListener('unhandledrejection', e => {
    e.preventDefault();
    const err = e.reason;
    report({
      type: err?.message?.includes('Loading chunk') ? 'chunk_error' : 'promise_error',
      msg: err?.message || String(err),
      stack: err?.stack
    });
  });

  // 3. 资源加载失败
  window.addEventListener('error', e => {
    const tag = e.target?.tagName;
    if (['IMG','SCRIPT','LINK','VIDEO'].includes(tag)) {
      report({
        type: 'resource_error',
        tag,
        src: e.target.src || e.target.href
      });
    }
  }, true);

  // 4. 白屏检测
  function checkWhite() {
    const root = document.querySelector('#app,#root');
    if (!root || root.children.length === 0) {
      report({ type: 'white_screen', html: document.body.innerHTML.slice(0, 600) });
    }
  }
  setTimeout(checkWhite, 3000);
  setTimeout(checkWhite, 6000);

  // 5. FCP慢渲染监控
  if (window.PerformanceObserver) {
    const obs = new PerformanceObserver(list => {
      list.getEntries().forEach(entry => {
        if (entry.name === 'first-contentful-paint' && entry.startTime > 3000) {
          report({ type: 'slow_fcp', fcp: entry.startTime });
        }
      });
    });
    obs.observe({ type: 'paint', buffered: true });
  }
})();
</script>
```

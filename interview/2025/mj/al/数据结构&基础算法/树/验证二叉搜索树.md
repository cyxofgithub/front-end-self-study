## 验证二叉搜索树

思路 1：

-   预设一个 pre = -Infinity 通过中序遍历比对前一个值是否小于或等于当前值，如果不是说明不是二叉搜索树

思路 2:

-   与思路 1 一样，换成栈来模拟递归过程

```javascript
// 口诀举例：
// - "压左路到底，弹出节点处理，切右再走起！"
// 栈方式模拟递归的遍历基本框架
function traverseByStack(root) {
    const stack = [];
    let cur = root;
    while (stack.length > 0 || cur) {
        while (cur) {
            // 前序遍历位置：console.log(cur.val);
            stack.push(cur);
            cur = cur.left;
        }
        cur = stack.pop();
        // 中序遍历位置：console.log(cur.val);
        cur = cur.right;
    }
}
```

```javascript
// 后序遍历（栈模拟递归）的基本框架
function postOrderByStack(root) {
    const stack = [];
    let cur = root;
    let lastVisited = null;

    while (stack.length > 0 || cur) {
        while (cur) {
            stack.push(cur);
            cur = cur.left;
        }
        cur = stack[stack.length - 1];
        // 如果右子树空或已访问，则访问当前节点
        if (!cur.right || cur.right === lastVisited) {
            // 后序遍历位置：console.log(cur.val);
            stack.pop();
            lastVisited = cur;
            cur = null;
        } else {
            cur = cur.right;
        }
    }
}
```

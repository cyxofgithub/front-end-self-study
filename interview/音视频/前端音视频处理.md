## 一、ffmpeg.wasm 核心功能示例

实现 **视频裁剪**、**提取音频**、**格式转换** 三个高频功能，先安装/引入依赖：

```html
<!DOCTYPE html>
<html>
    <body>
        <input type="file" id="fileInput" accept="video/*,audio/*" />
        <button onclick="cropVideo()">裁剪视频</button>
        <button onclick="extractAudio()">提取音频</button>
        <button onclick="convertFormat()">转WebM格式</button>

        <!-- 引入 ffmpeg.wasm 核心库（CDN 版本） -->
        <script src="https://unpkg.com/@ffmpeg/ffmpeg@0.12.6/dist/ffmpeg.min.js"></script>
        <script>
            // 初始化 ffmpeg 实例
            const { createFFmpeg, fetchFile } = FFmpeg;
            const ffmpeg = createFFmpeg({ log: true }); // log: true 可查看处理日志

            // 全局存储文件信息
            let fileData = null;
            document.getElementById('fileInput').addEventListener('change', async (e) => {
                fileData = e.target.files[0];
                if (!ffmpeg.isLoaded()) await ffmpeg.load(); // 首次使用加载 wasm 核心
            });

            // 1. 视频裁剪：从 第10秒 开始，裁剪 5秒
            async function cropVideo() {
                if (!fileData) return alert('请先选择文件');

                // 1. 将文件写入 ffmpeg 虚拟文件系统
                ffmpeg.FS('writeFile', 'input.mp4', await fetchFile(fileData));

                // 2. 执行裁剪命令（-ss 开始时间，-t 持续时间，-c copy 快速拷贝编码，不重新编码）
                await ffmpeg.run(
                    '-i',
                    'input.mp4', // 输入文件
                    '-ss',
                    '00:00:10', // 开始时间
                    '-t',
                    '00:00:05', // 持续时间
                    '-c',
                    'copy', // 快速模式（不转码）
                    'output_crop.mp4' // 输出文件
                );

                // 3. 读取输出文件并下载
                const data = ffmpeg.FS('readFile', 'output_crop.mp4');
                downloadFile(data, '裁剪后的视频.mp4', 'video/mp4');
            }

            // 2. 提取音频：从视频中提取 MP3 音频
            async function extractAudio() {
                if (!fileData) return alert('请先选择文件');

                ffmpeg.FS('writeFile', 'input.mp4', await fetchFile(fileData));
                // 提取音频流，转码为 MP3（-vn 禁用视频，-acodec mp3 指定音频编码）
                await ffmpeg.run(
                    '-i',
                    'input.mp4',
                    '-vn', // 只保留音频
                    '-acodec',
                    'mp3', // 音频编码格式
                    'output_audio.mp3'
                );

                const data = ffmpeg.FS('readFile', 'output_audio.mp3');
                downloadFile(data, '提取的音频.mp3', 'audio/mp3');
            }

            // 3. 格式转换：MP4 转 WebM（适合网页播放）
            async function convertFormat() {
                if (!fileData) return alert('请先选择文件');

                ffmpeg.FS('writeFile', 'input.mp4', await fetchFile(fileData));
                // 转码为 WebM（VP9 编码，体积更小）
                await ffmpeg.run(
                    '-i',
                    'input.mp4',
                    '-c:v',
                    'vp9', // 视频编码 VP9
                    '-c:a',
                    'opus', // 音频编码 Opus
                    'output.webm'
                );

                const data = ffmpeg.FS('readFile', 'output.webm');
                downloadFile(data, '转换后的视频.webm', 'video/webm');
            }

            // 通用下载函数
            function downloadFile(data, fileName, mimeType) {
                const url = URL.createObjectURL(new Blob([data.buffer], { type: mimeType }));
                const a = document.createElement('a');
                a.href = url;
                a.download = fileName;
                a.click();
                URL.revokeObjectURL(url); // 释放内存
            }
        </script>
    </body>
</html>
```

**关键说明**：

-   `ffmpeg.run()` 中的参数和原生 FFmpeg 命令完全一致，可参考 FFmpeg 官方文档扩展功能（如合并视频需先转码为相同编码，再用 `concat` 协议）。
-   `-c copy` 是“快速模式”，只裁剪容器层，不重新编码，速度极快；若需要调整分辨率/编码，需去掉该参数并指定编码格式。

## 二、视频帧处理示例

实现 **截取指定帧**、**实时逐帧分析**、**简单滤镜** 三个功能：

```html
<!DOCTYPE html>
<html>
    <body>
        <input type="file" id="videoInput" accept="video/*" />
        <video id="video" controls style="display: none;"></video>
        <canvas id="frameCanvas" width="640" height="360"></canvas>
        <button onclick="captureFrame()">截取第5秒帧</button>
        <button onclick="analyzeFrames()">逐帧分析（10帧）</button>
        <button onclick="addFilter()">添加灰度滤镜</button>

        <script>
            const video = document.getElementById('video');
            const canvas = document.getElementById('frameCanvas');
            const ctx = canvas.getContext('2d');
            let videoFile = null;

            // 加载视频文件
            document.getElementById('videoInput').addEventListener('change', (e) => {
                videoFile = e.target.files[0];
                video.src = URL.createObjectURL(videoFile);
                video.load();
            });

            // 1. 截取指定时间的帧（第5秒）
            function captureFrame() {
                if (!videoFile) return alert('请选择视频');

                // 定位到第5秒
                video.currentTime = 5;
                // 等待视频帧加载完成
                video.onseeked = () => {
                    // 将视频帧绘制到 Canvas
                    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                    // 可选：下载帧图片
                    const dataUrl = canvas.toDataURL('image/png');
                    const a = document.createElement('a');
                    a.href = dataUrl;
                    a.download = '第5秒帧.png';
                    a.click();
                };
            }

            // 2. 逐帧分析（读取前10帧的像素数据）
            async function analyzeFrames() {
                if (!videoFile) return alert('请选择视频');

                video.currentTime = 0;
                let frameCount = 0;
                const totalFrames = 10;

                // 逐帧读取
                function readFrame() {
                    if (frameCount >= totalFrames) return;

                    // 绘制当前帧
                    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                    // 获取像素数据（Uint8Array，每个像素 RGBA 4个值）
                    const pixelData = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
                    // 简单分析：计算平均亮度
                    let totalBrightness = 0;
                    for (let i = 0; i < pixelData.length; i += 4) {
                        const r = pixelData[i];
                        const g = pixelData[i + 1];
                        const b = pixelData[i + 2];
                        totalBrightness += (r + g + b) / 3; // 灰度值 = 亮度
                    }
                    const avgBrightness = totalBrightness / (canvas.width * canvas.height);
                    console.log(`第${frameCount + 1}帧 平均亮度：`, avgBrightness);

                    frameCount++;
                    // 前进一帧（约 1/30 秒，适配 30fps 视频）
                    video.currentTime += 1 / 30;
                    requestAnimationFrame(readFrame);
                }

                // 等待视频加载完成后开始分析
                video.onloadeddata = readFrame;
            }

            // 3. 添加灰度滤镜（Canvas 实现）
            function addFilter() {
                if (!videoFile) return alert('请选择视频');

                video.currentTime = 5;
                video.onseeked = () => {
                    // 1. 绘制原始帧
                    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                    // 2. 获取像素数据
                    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                    const data = imageData.data;
                    // 3. 灰度处理：R=G=B=(R+G+B)/3
                    for (let i = 0; i < data.length; i += 4) {
                        const gray = (data[i] + data[i + 1] + data[i + 2]) / 3;
                        data[i] = gray; // R
                        data[i + 1] = gray; // G
                        data[i + 2] = gray; // B
                        // A 通道不变
                    }
                    // 4. 绘制处理后的帧
                    ctx.putImageData(imageData, 0, 0);
                };
            }
        </script>
    </body>
</html>
```

**关键说明**：

-   逐帧分析可结合 TensorFlow.js 实现 AI 识别（如人脸检测：加载 `blazeface` 模型，对每帧像素数据推理）。
-   高性能滤镜建议用 WebGL 着色器（Canvas 像素操作适合简单场景，大分辨率下卡顿）。

## 三、音频处理示例（Web Audio API）

实现 **音量增益**、**简单降噪**、**回声消除** 核心功能：

```html
<!DOCTYPE html>
<html>
    <body>
        <input type="file" id="audioInput" accept="audio/*" />
        <button onclick="adjustVolume()">音量+2倍</button>
        <button onclick="reduceNoise()">简单降噪</button>
        <button onclick="echoCancellation()">回声消除（需浏览器支持）</button>

        <script>
            let audioContext = null;
            let audioBuffer = null;

            // 加载音频文件
            document.getElementById('audioInput').addEventListener('change', async (e) => {
                const file = e.target.files[0];
                if (!file) return;

                // 初始化音频上下文（需用户交互触发，浏览器安全限制）
                audioContext = audioContext || new AudioContext();
                // 解码音频文件为 AudioBuffer
                const arrayBuffer = await file.arrayBuffer();
                audioBuffer = await audioContext.decodeAudioData(arrayBuffer);
            });

            // 1. 音量增益（2倍）
            function adjustVolume() {
                if (!audioBuffer || !audioContext) return alert('请选择音频文件');

                // 创建增益节点
                const gainNode = audioContext.createGain();
                gainNode.gain.value = 2; // 2倍音量（0.5=减半，0=静音）

                // 连接音频链路：源 → 增益 → 输出
                const source = audioContext.createBufferSource();
                source.buffer = audioBuffer;
                source.connect(gainNode);
                gainNode.connect(audioContext.destination);

                // 播放处理后的音频
                source.start();
            }

            // 2. 简单降噪（噪声门：低于阈值的音频设为0）
            async function reduceNoise() {
                if (!audioBuffer || !audioContext) return alert('请选择音频文件');

                // 复制音频数据（避免修改原数据）
                const newBuffer = audioContext.createBuffer(
                    audioBuffer.numberOfChannels,
                    audioBuffer.length,
                    audioBuffer.sampleRate
                );

                // 遍历所有声道和样本点
                for (let channel = 0; channel < audioBuffer.numberOfChannels; channel++) {
                    const channelData = audioBuffer.getChannelData(channel);
                    const newChannelData = newBuffer.getChannelData(channel);

                    // 噪声阈值（可根据实际音频调整，0.01 对应 -40dB）
                    const threshold = 0.01;
                    for (let i = 0; i < channelData.length; i++) {
                        // 低于阈值则静音，高于则保留
                        newChannelData[i] = Math.abs(channelData[i]) < threshold ? 0 : channelData[i];
                    }
                }

                // 播放降噪后的音频
                const source = audioContext.createBufferSource();
                source.buffer = newBuffer;
                source.connect(audioContext.destination);
                source.start();
            }

            // 3. 回声消除（依赖浏览器的 EchoCancellation 特性）
            function echoCancellation() {
                if (!navigator.mediaDevices) return alert('浏览器不支持媒体设备');

                // 申请麦克风权限，并开启回声消除
                navigator.mediaDevices
                    .getUserMedia({
                        audio: {
                            echoCancellation: true, // 核心：开启回声消除
                            noiseSuppression: true, // 顺带开启降噪
                            autoGainControl: true, // 自动增益
                        },
                    })
                    .then((stream) => {
                        // 播放处理后的麦克风音频
                        const source = audioContext.createMediaStreamSource(stream);
                        source.connect(audioContext.destination);
                    })
                    .catch((err) => {
                        console.error('获取麦克风失败：', err);
                        alert('需授予麦克风权限才能测试回声消除');
                    });
            }
        </script>
    </body>
</html>
```

**关键说明**：

-   高级降噪（如 RNNoise）需引入 WebAssembly 版本的 RNNoise 库，示例中是极简的“噪声门”方案，适合入门。
-   回声消除仅对麦克风输入有效，且依赖浏览器/设备支持（桌面端 Chrome 支持较好，移动端部分机型不支持）。

### 总结

1. **ffmpeg.wasm** 是前端音视频处理的“万能工具”，核心是通过 `ffmpeg.run()` 执行原生 FFmpeg 命令，需注意加载 wasm 的耗时和内存占用。
2. **视频帧处理** 基于 Canvas/WebCodecs，截取帧用 `drawImage`，逐帧分析需循环读取 `currentTime`，滤镜优先用 WebGL 提升性能。
3. **音频处理** 核心是 Web Audio API 的“节点链路”（源 → 处理节点 → 输出），简单降噪用噪声门，高级效果需结合 WASM 库。
